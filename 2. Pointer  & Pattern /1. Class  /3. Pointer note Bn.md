## ১. মেমরি বেসিকস এবং অ্যাড্রেস-অব অপারেটর (&)
আমরা যখনই কোনো ভ্যারিয়েবল ডিক্লেয়ার করি, সেটি কম্পিউটারের র‍্যামের (RAM) একটি নির্দিষ্ট স্লটে জমা হয়। প্রতিটি স্লটের একটি ইউনিক হেক্সাডেসিমেল (Hexadecimal) অ্যাড্রেস বা ঠিকানা থাকে।

* & (Address-of operator): এটি কোনো ভ্যারিয়েবলের মেমরি অ্যাড্রেস খুঁজে বের করতে ব্যবহৃত হয়।

#include <iostream>using namespace std;
int main() {
    int age = 22;
    
    cout << "Value of age: " << age << endl; // আউটপুট: 22
    cout << "Memory Address of age: " << &age << endl; // আউটপুট: 0x7ffee3b44b8c (নমুনা হেক্স অ্যাড্রেস)
    
    return 0;
}

## ২. পয়েন্টার কী? (ডিক্লেয়ারেশন এবং ডিরেফারেন্সিং)
পয়েন্টার হলো একটি বিশেষ ধরণের ভ্যারিয়েবল যা অন্য একটি ভ্যারিয়েবলের মেমরি অ্যাড্রেস নিজের মধ্যে জমা বা স্টোর করে রাখে।

* ഡিক্লেয়ারেশন সিনট্যাক্স: data_type* pointer_name;
* ডিরেফারেন্সিং (* অপারেটর): পয়েন্টারটি যে মেমরি অ্যাড্রেসকে নির্দেশ করছে, সেই ঠিকানায় থাকা ভ্যালু দেখা বা পরিবর্তন করার পদ্ধতি।

#include <iostream>using namespace std;
int main() {
    int score = 100;
    int* ptr = &score; // 'ptr' নামের পয়েন্টারটি 'score' এর অ্যাড্রেস স্টোর করছে

    cout << "Address stored in ptr: " << ptr << endl;   // &score এর সমান অ্যাড্রেস দেখাবে
    cout << "Value pointed to by ptr: " << *ptr << endl; // ডিরেফারেন্সিং: আউটপুট আসবে 100

    // পয়েন্টার এর মাধ্যমে ভ্যালু পরিবর্তন (Modifying value)
    *ptr = 150; 
    cout << "New value of score: " << score << endl;    // আউটপুট আসবে 150
    
    return 0;
}

## ৩. নাল পয়েন্টার, ওয়াইল্ড পয়েন্টার এবং ভয়েড পয়েন্টার## ওয়াইল্ড পয়েন্টার (Wild Pointer)
যে পয়েন্টার ডিক্লেয়ার করা হয়েছে কিন্তু কোনো ভ্যালু দিয়ে ইনিশিয়ালইজ করা হয়নি। এটি মেমরির যেকোনো র্যান্ডম বা গার্বেজ লোকেশনকে পয়েন্ট করে থাকে। এটি ব্যবহার করলে কোড ক্র্যাশ (Segmentation Fault) হতে পারে।

int* wild_ptr; // বিপজ্জনক! এটি গার্বেজ অ্যাড্রেস পয়েন্ট করে আছে।

## নাল পয়েন্টার (Null Pointer)
যে পয়েন্টারটি সুনির্দিষ্টভাবে কোনো কিছুকেই পয়েন্ট করে না। মডার্ন সি++ (C++11 এবং পরবর্তী সংস্করণ) এ এর জন্য nullptr ব্যবহার করা হয়। পয়েন্টার ব্যবহারের আগে এটি নাল কি না তা চেক করা ভালো অভ্যাস।

int* null_ptr = nullptr; // নিরাপদ ইনিশিয়ালাইজেশন
if (null_ptr != nullptr) {
    cout << *null_ptr;
}

## ভয়েড পয়েন্টার / জেনেরিক পয়েন্টার (Void Pointer)
একটি বিশেষ ধরণের পয়েন্টার (void*) যা যেকোনো ডেটা টাইপের ভ্যারিয়েবলের অ্যাড্রেস হোল্ড করতে পারে। তবে এটিকে সরাসরি ডিরেফারেন্স করা যায় না, প্রথমে টাইপকাস্ট (Typecast) করে নিতে হয়।

#include <iostream>using namespace std;
int main() {
    int num = 10;
    double d = 5.5;
    
    void* generic_ptr;
    
    generic_ptr = &num; // int কে পয়েন্ট করছে
    // cout << *generic_ptr; // ❌ ভুল! void* সরাসরি ডিরেফারেন্স করা সম্ভব নয়।
    cout << "Int value via void*: " << *(static_cast<int*>(generic_ptr)) << endl; // প্রথমে টাইপকাস্ট করা হলো

    generic_ptr = &d;   // double কে পয়েন্ট করছে
    cout << "Double value via void*: " << *(static_cast<double*>(generic_ptr)) << endl;
    
    return 0;
}

## ৪. পয়েন্টার অ্যারিথমেটিক এবং অ্যারে (লুকানো সংযোগ)
একটি অ্যারের নাম মূলত একটি পয়েন্টার হিসেবে কাজ করে, যা অ্যারের প্রথম উপাদানটিকে (index 0) নির্দেশ করে।
যখন আপনি পয়েন্টার অ্যারিথমেটিক করবেন (যেমন ptr + 1), কম্পাইলার সেই অ্যাড্রেসকে 1 * sizeof(data_type) বাইট বাড়িয়ে দেয়।

#include <iostream>using namespace std;
int main() {
    int arr[3] = {10, 20, 30};
    int* ptr = arr; // এখানে arr এবং &arr[0] একই কথা

    cout << "Address of arr[0]: " << ptr << " | Value: " << *ptr << endl;
    cout << "Address of arr[1]: " << (ptr + 1) << " | Value: " << *(ptr + 1) << endl; // ৪ বাইট সামনে এগিয়ে যাবে (যেহেতু int = 4 bytes)
    cout << "Address of arr[2]: " << (ptr + 2) << " | Value: " << *(ptr + 2) << endl;

    // অ্যারে নোটেশন ব্যাকএন্ডে এভাবে কাজ করে: arr[i] আসলে কম্পাইল হয় *(arr + i) হিসেবে
    cout << "Using array notation equivalence: " << arr[1] << " == " << *(arr + 1) << endl;
    return 0;
}

## ৫. পাস বাই ভ্যালু বনাম পাস বাই রেফারেন্স বনাম পাস বাই পয়েন্টার

| বৈশিষ্ট্য (Feature) | পাস বাই ভ্যালু (Pass by Value) | পাস বাই রেফারেন্স (Pass by Reference) | পাস বাই পয়েন্টার (Pass by Pointer) |
|---|---|---|---|
| ফাংশনে কী পাস হয়? | ভ্যারিয়েবলের একটি কপি (Copy) | ভ্যারিয়েবলের একটি উপনাম (Alias) | ভ্যারিয়েবলের মেমরি অ্যাড্রেস |
| সিনট্যাক্স (Syntax) | void func(int x) | void func(int &x) | void func(int *x) |
| মূল ভ্যালু পরিবর্তন হয়? | না | হ্যাঁ | |

#include <iostream>using namespace std;
void modifyByValue(int x) { x = 99; }void modifyByReference(int &x) { x = 99; }void modifyByPointer(int *x) { *x = 100; }
int main() {
    int num = 10;

    modifyByValue(num);
    cout << "After Pass by Value: " << num << endl; // আউটপুট: 10 (কোনো পরিবর্তন হয়নি)

    modifyByReference(num);
    cout << "After Pass by Reference: " << num << endl; // আউটপুট: 99 (পরিবর্তন হয়েছে)

    modifyByPointer(&num);
    cout << "After Pass by Pointer: " << num << endl; // আউটপুট: 100 (পরিবর্তন হয়েছে)

    return 0;
}

## ৬. কনস্ট-পয়েন্টার ম্যাট্রিক্স
const এবং পয়েন্টার একসাথে কীভাবে কাজ করে তা বোঝার সহজ ম্যাপিং নিচে দেওয়া হলো:

   1. পয়েন্টার টু কনস্ট্যান্ট ভ্যালু (const int* ptr)
   * পয়েন্টারটি যে ভ্যালু নির্দেশ করছে তা পরিবর্তন করা যাবে না। তবে পয়েন্টারে থাকা মেমরি অ্যাড্রেস পরিবর্তন করা যাবে।
   2. কনস্ট্যান্ট পয়েন্টার (int* const ptr)
   * পয়েন্টারে থাকা মেমরি অ্যাড্রেস পরিবর্তন করা যাবে না। তবে সেই অ্যাড্রেসে থাকা ভ্যালু পরিবর্তন করা যাবে।
   3. কনস্ট্যান্ট পয়েন্টার টু কনস্ট্যান্ট ভ্যালু (const int* const ptr)
   * এখানে ভ্যালু এবং মেমরি অ্যাড্রেস কোনোটিই পরিবর্তন করা সম্ভব নয়।
   
#include <iostream>using namespace std;
int main() {
    int a = 10;
    int b = 20;
    const int c = 30;

    // 1. Pointer to Constant
    const int* ptr1 = &c;
    // *ptr1 = 40;     // ❌ ভুল! ভ্যালু পরিবর্তন করা যাবে না (Constant)।
    ptr1 = &a;         //  সঠিক! অন্য অ্যাড্রেস ট্র্যাক করা যাবে।

    // 2. Constant Pointer
    int* const ptr2 = &a;
    *ptr2 = 40;        //  সঠিক! ভ্যালু পরিবর্তন করা যাবে।
    // ptr2 = &b;      // ❌ ভুল! অ্যাড্রেস পরিবর্তন করা যাবে না (Constant)।

    // 3. Constant Pointer to Constant
    const int* const ptr3 = &c;
    // *ptr3 = 50;     // ❌ ভুল!
    // ptr3 = &b;      // ❌ ভুল!

    return 0;
}

## ৭. ডায়নামিক মেমরি অ্যালোকেশন (1D and 2D Arrays)
র‍্যামের Heap মেমরিতে যখন new কিওয়ার্ড দিয়ে মেমরি বরাদ্দ করা হয়, তখন কাজ শেষে delete বা delete[] দিয়ে তা ম্যানুয়ালি খালি করতে হয়। তা না করলে মেমরি লিক (Memory Leak) ঘটে।
## ১ডি ডায়নামিক অ্যারে (1D Dynamic Array)

#include <iostream>using namespace std;
int main() {
    int size = 3;
    int* arr = new int[size]; // হিপ (Heap) মেমরিতে বরাদ্দ করা হলো

    // ভ্যালু অ্যাসাইন করা হচ্ছে
    arr[0] = 5; arr[1] = 10; arr[2] = 15;

    for(int i = 0; i < size; i++) {
        cout << arr[i] << " ";
    }
    cout << endl;

    delete[] arr;  // ⚠️ অত্যন্ত জরুরি: মেমরি ব্লক খালি করা হলো
    arr = nullptr; // ড্যাঙ্গলিং পয়েন্টার এড়াতে নাল করে দেওয়া হলো
    return 0;
}

## ২ডি ডায়নামিক অ্যারে (2D Dynamic Array)
ডায়নামিকালি একটি 2D অ্যারে তৈরি করতে প্রথমে পয়েন্টারদের একটি অ্যারে (Array of Pointers) তৈরি করতে হয় এবং তারপর লুপ চালিয়ে প্রতিটি পয়েন্টারের জন্য আলাদা এক একটি অ্যারে বরাদ্দ করতে হয়।

#include <iostream>using namespace std;
int main() {
    int rows = 3, cols = 4;

    // ধাপ ১: পয়েন্টারদের একটি অ্যারে তৈরি (রো বা সারির জন্য)
    int** matrix = new int*[rows];

    // ধাপ ২: প্রতিটি সারির অধীনে কলাম বা আসল ভ্যালুর জন্য মেমরি বরাদ্দ
    for (int i = 0; i < rows; i++) {
        matrix[i] = new int[cols];
    }

    // টেস্ট ভ্যালু বসানো
    matrix[0][0] = 42;
    cout << "Value at matrix[0][0]: " << matrix[0][0] << endl;

    // ⚠️ ধাপ ৩: মেমরি খালি করার সময় উল্টো ক্রমে করতে হবে
    for (int i = 0; i < rows; i++) {
        delete[] matrix[i]; // প্রথমে ভেতরের সারিগুলো ডিলিট করা হলো
    }
    delete[] matrix;        // সবশেষে মেইন ডাবল পয়েন্টারটি ডিলিট করা হলো
    matrix = nullptr;
    
    return 0;
}

## ৮. অ্যাডভান্সড পয়েন্টার কনসেপ্ট## পয়েন্টার টু পয়েন্টার / ডাবল পয়েন্টার (Pointer to Pointer)
একটি ডাবল পয়েন্টার অন্য একটি পয়েন্টারের মেমরি অ্যাড্রেস স্টোর করে।

int var = 500;int* ptr = &var;   // সিঙ্গেল পয়েন্টারint** dptr = &ptr; // ডাবল পয়েন্টার (ptr এর অ্যাড্রেস হোল্ড করছে)

cout << "Value via dptr: " << **dptr; // আউটপুট: 500

## ফাংশন পয়েন্টার (Function Pointers)
ফাংশন পয়েন্টার হলো এমন একটি পয়েন্টার যা কোনো ভ্যারিয়েবলের বদলে মেমরিতে থাকা একটি ফাংশনের এক্সিকিউটেবল কোডের অ্যাড্রেস স্টোর করে। এর মাধ্যমে রানটাইমে ডায়নামিকালি ফাংশন কল করা যায়।

#include <iostream>using namespace std;
void greet(int count) {
    cout << "Hello, Count is: " << count << endl;
}
int main() {
    // ফাংশন পয়েন্টার ডিক্লেয়ারেশন এবং অ্যাসাইনমেন্ট
    void (*funcPtr)(int) = &greet; 
    
    // পয়েন্টার এর মাধ্যমে ফাংশন কল
    funcPtr(5); 
    return 0;
}

## অবজেক্ট ওরিয়েন্টেড প্রোগ্রামিং-এ this পয়েন্টার
সি++ এ ক্লাসের ভেতর প্রতিটি মেম্বার ফাংশনে একটি ইনবিল্ট বা সুপ্ত পয়েন্টার থাকে যার নাম this। এটি ফাংশনটি কল করা বর্তমান অবজেক্টের নিজস্ব অ্যাড্রেস নির্দেশ করে। লোকাল ভ্যারিয়েবল এবং ক্লাসের মেম্বার ভ্যারিয়েবলের নাম একই হলে এটি কনফিউশন দূর করে।

#include <iostream>using namespace std;
class Player {private:
    int health;public:
    Player(int health) {
        // health = health; // ❌ ভুল! ক্লাসের ভ্যারিয়েবল নাকি প্যারামিটার তা কম্পাইলার বুঝবে না।
        this->health = health; // ✔️ সঠিক! this->health ক্লাসের মেম্বার ভ্যারিয়েবলকে বোঝাচ্ছে।
    }
    void showHealth() { cout << "Health: " << this->health << endl; }
};
int main() {
    Player p1(100);
    p1.showHealth();
    return 0;
}

## স্মার্ট পয়েন্টার / মডার্ন সি++ (Smart Pointers)
ম্যানুয়াল মেমরি ম্যানেজমেন্ট এবং delete ব্যবহারের ঝামেলা থেকে মুক্তি পেতে C++11 এ Smart Pointers নিয়ে আসা হয়। এগুলো মেমরি স্কোপের বাইরে চলে গেলে স্বয়ংক্রিয়ভাবে মেমরি রিলিজ বা ডিলিট করে দেয়। এর জন্য <memory> হেডার ফাইল ব্যবহার করতে হয়।
১. unique_ptr: মেমরির একটি ব্লকের ওপর শুধুমাত্র একজন মালিক বা ওনার (Exclusive ownership) থাকতে পারে। এটি কপি করা যায় না।
২. shared_ptr: একই মেমরি লোকেশন একাধিক পয়েন্টার শেয়ার করতে পারে। একটি ইন্টারনাল রেফারেন্স কাউন্টারের মাধ্যমে এটি ট্র্যাক রাখা হয়।

#include <iostream>#include <memory> // স্মার্ট পয়েন্টার এর জন্য আবশ্যকusing namespace std;
class Widget {public:
    Widget() { cout << "Widget Created\n"; }
    ~Widget() { cout << "Widget Destroyed Automatically\n"; }
};
int main() {
    // scope block
    {
        unique_ptr<Widget> w1 = make_unique<Widget>(); 
        // এখানে ম্যানুয়ালি delete w1 লেখার কোনো প্রয়োজন নেই। 
    } // এই ব্র্যাকেটে এসে w1 স্কোপের বাইরে যাওয়ার সাথে সাথে মেমরি নিজে থেকেই মুক্ত হয়ে যাবে।
    
    return 0;
}
